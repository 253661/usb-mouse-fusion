cmake_minimum_required(VERSION 3.15)
get_filename_component(FUSION_REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(FUSION_WORKSPACE "${FUSION_REPO}/.." ABSOLUTE)
set(PICO_SDK_PATH "${FUSION_REPO}/.build-deps/pico-sdk" CACHE PATH "Pinned existing SDK")
set(PICO_TOOLCHAIN_PATH "${FUSION_REPO}/.build-deps/arm" CACHE PATH "Existing ARM GCC")
set(picotool_DIR "${FUSION_WORKSPACE}/.rp2040-tools/picotool/picotool" CACHE PATH "Existing build conversion tool only")
set(PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/../cdc_alive/boards")
set(PICO_BOARD c4_l4 CACHE STRING "Current native board")
set(PICO_PLATFORM rp2040 CACHE STRING "Current MCU")
# Must precede SDK initialization. The SDK/TinyUSB integration adds vendor/HCD exactly once.
set(PICO_PIO_USB_PATH "${CMAKE_CURRENT_LIST_DIR}/../rp2040/vendor/pico-pio-usb")
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(c4_l4_mouse_passthrough C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(FUSION_DIAGNOSTIC_CDC "Expose bounded diagnostics in private RAM test image" ON)
option(FUSION_CONTROLLED_J2 "Explicit POWER/ARM test commands; still defaults OFF" OFF)
option(FUSION_STANDALONE_FLASH "HID-only Flash boot with interlocked automatic J2 startup" OFF)
option(FUSION_J4_LINK "Isolated UART0 information/control session; no auxiliary motion" OFF)
option(FUSION_J4_AUX "Capability-gated RAM modes/config and Core0 X/Y ingress" OFF)
option(FUSION_TRAJECTORY "Offline candidate: independent MFT1 runtime/triggers; no real Flash writer" OFF)
option(FUSION_PERSISTENCE "Explicit maintenance Flash save and boot preset restore" OFF)
if(FUSION_PERSISTENCE AND NOT FUSION_TRAJECTORY)
    message(FATAL_ERROR "Persistent presets require trajectory runtime")
endif()
set(FUSION_USB_IDENTITY_CONFIG "${CMAKE_CURRENT_LIST_DIR}/usb_identity/board-001.json" CACHE FILEPATH "Persistent board USB identity JSON")
set(FUSION_USB_IDENTITY_HEADER "" CACHE FILEPATH "Legacy explicit header override; use JSON for reviewed deployment")
if(FUSION_TRAJECTORY AND NOT FUSION_J4_AUX)
    message(FATAL_ERROR "Trajectory runtime requires API auxiliary support")
endif()
if(FUSION_J4_AUX AND NOT FUSION_J4_LINK)
    message(FATAL_ERROR "API auxiliary ingress requires J4")
endif()
if(FUSION_J4_LINK AND NOT FUSION_STANDALONE_FLASH)
    message(FATAL_ERROR "J4 link requires the explicit Standalone Flash profile")
endif()
if(FUSION_STANDALONE_FLASH AND (FUSION_CONTROLLED_J2 OR FUSION_DIAGNOSTIC_CDC))
    message(FATAL_ERROR "Standalone Flash requires HID-only, separate from controlled RAM")
endif()
if(FUSION_STANDALONE_FLASH)
    # Match the board's already-tested CDC Flash bootloader explicitly.
    set(PICO_DEFAULT_BOOT_STAGE2_FILE "${PICO_SDK_PATH}/src/rp2040/boot_stage2/boot2_generic_03h.S")
endif()
if(FUSION_CONTROLLED_J2 AND NOT FUSION_DIAGNOSTIC_CDC)
    message(FATAL_ERROR "Controlled J2 tests require CDC; HID-only stays locked")
endif()
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING STREQUAL "2.2.0" OR NOT PICO_PLATFORM STREQUAL "rp2040" OR NOT PICO_BOARD STREQUAL "c4_l4")
    message(FATAL_ERROR "Requires pinned SDK 2.2.0, RP2040, C4-L4")
endif()
if(NOT TARGET tinyusb_pico_pio_usb)
    message(FATAL_ERROR "Current SDK did not add the pinned PIO USB integration")
endif()
include(vendor/host_session/build.cmake)
get_target_property(fusion_host_sources tinyusb_host_base INTERFACE_SOURCES)
set(fusion_host_replaced 0)
set(fusion_host_patched_sources "")
foreach(source IN LISTS fusion_host_sources)
    if(source MATCHES "/host/usbh\\.c$")
        list(APPEND fusion_host_patched_sources "${FUSION_USBH_SOURCE}")
        math(EXPR fusion_host_replaced "${fusion_host_replaced}+1")
    else()
        list(APPEND fusion_host_patched_sources "${source}")
    endif()
endforeach()
if(NOT fusion_host_replaced EQUAL 1)
    message(FATAL_ERROR "Expected one pinned USBH source in tinyusb_host_base")
endif()
set_property(TARGET tinyusb_host_base PROPERTY INTERFACE_SOURCES "${fusion_host_patched_sources}")
add_subdirectory(vendor/hid-report-parser)
add_executable(c4_l4_mouse_passthrough main.c board_io.c host_port.c diagnostics.c
    usb_descriptors.c hid_output.c hid_input.c hid_layout.cpp hid_decode.c hid_selection.c button_state.c mouse_report.c log_queue.c
    event_bus.c passthrough_core.c lifecycle.c bringup_control.c host_wait.c host_pio_state.c host_commands.c)
if(FUSION_J4_LINK)
    target_sources(c4_l4_mouse_passthrough PRIVATE j4_link.c j4_uart.c)
    target_link_libraries(c4_l4_mouse_passthrough PRIVATE hardware_uart pico_unique_id)
    target_link_options(c4_l4_mouse_passthrough PRIVATE -Wl,--undefined=fusion_j4_contract)
endif()
if(FUSION_J4_AUX)
    target_sources(c4_l4_mouse_passthrough PRIVATE j4_aux_bridge.c ai_smooth.c)
    target_link_options(c4_l4_mouse_passthrough PRIVATE -Wl,--undefined=fusion_api_contract -Wl,--undefined=fusion_ai_gate_contract -Wl,--undefined=fusion_ai_queue_contract -Wl,--undefined=fusion_ai_smooth_contract)
endif()
if(FUSION_TRAJECTORY)
    target_sources(c4_l4_mouse_passthrough PRIVATE trajectory.c preset_runtime.c preset_rule.c preset_motion.c preset_perturb.c
        preset_trigger.c preset_store.c preset_persistence.c preset_flash_backend.c preset_flash_pico.c j4_preset.c)
    target_link_options(c4_l4_mouse_passthrough PRIVATE -Wl,--undefined=fusion_preset_contract -Wl,--undefined=preset_flash_pico_boot_init -Wl,--undefined=fusion_preset_flash_contract
        "-Wl,-T,${CMAKE_CURRENT_LIST_DIR}/preset_reservation.ld")
endif()
if(FUSION_STANDALONE_FLASH)
    pico_set_binary_type(c4_l4_mouse_passthrough copy_to_ram)
    target_link_options(c4_l4_mouse_passthrough PRIVATE -Wl,--undefined=fusion_flash_contract)
else()
    pico_set_binary_type(c4_l4_mouse_passthrough no_flash)
endif()
target_include_directories(c4_l4_mouse_passthrough PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${PICO_SDK_PATH}/lib/tinyusb/src/host)
target_compile_definitions(c4_l4_mouse_passthrough PRIVATE FUSION_DIAGNOSTIC_CDC=$<BOOL:${FUSION_DIAGNOSTIC_CDC}> FUSION_CONTROLLED_J2=$<BOOL:${FUSION_CONTROLLED_J2}> FUSION_STANDALONE_FLASH=$<BOOL:${FUSION_STANDALONE_FLASH}> PICO_CORE1_STACK_SIZE=4096)
target_compile_definitions(c4_l4_mouse_passthrough PRIVATE FUSION_J4_LINK=$<BOOL:${FUSION_J4_LINK}>)
target_compile_definitions(c4_l4_mouse_passthrough PRIVATE FUSION_J4_AUX=$<BOOL:${FUSION_J4_AUX}>)
target_compile_definitions(c4_l4_mouse_passthrough PRIVATE FUSION_TRAJECTORY=$<BOOL:${FUSION_TRAJECTORY}>)
target_compile_definitions(c4_l4_mouse_passthrough PRIVATE FUSION_PERSISTENCE=$<BOOL:${FUSION_PERSISTENCE}>)
if(FUSION_PERSISTENCE)
    target_compile_definitions(c4_l4_mouse_passthrough PRIVATE PICO_STACK_SIZE=4096)
    target_link_options(c4_l4_mouse_passthrough PRIVATE -Wl,--undefined=fusion_persistence_contract)
endif()
if(NOT FUSION_USB_IDENTITY_HEADER)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/identity_profile.py"
        --config "${FUSION_USB_IDENTITY_CONFIG}" --build "${CMAKE_CURRENT_BINARY_DIR}"
        RESULT_VARIABLE fusion_identity_result)
    if(NOT fusion_identity_result EQUAL 0)
        message(FATAL_ERROR "USB identity snapshot failed; use a new build directory")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${FUSION_USB_IDENTITY_CONFIG}")
    set(FUSION_USB_IDENTITY_HEADER "${CMAKE_CURRENT_BINARY_DIR}/usb_identity_generated.h")
endif()
if(FUSION_USB_IDENTITY_HEADER)
    if(NOT EXISTS "${FUSION_USB_IDENTITY_HEADER}")
        message(FATAL_ERROR "Generate USB identity header with software/usb_identity.py first")
    endif()
    get_filename_component(fusion_identity_header "${FUSION_USB_IDENTITY_HEADER}" ABSOLUTE)
    target_compile_options(c4_l4_mouse_passthrough PRIVATE -include "${fusion_identity_header}")
endif()
target_link_libraries(c4_l4_mouse_passthrough PRIVATE pico_stdlib pico_multicore hardware_clocks
    hardware_dma hardware_pio tinyusb_device tinyusb_host tinyusb_pico_pio_usb fusion_hid_parser)
target_compile_options(c4_l4_mouse_passthrough PRIVATE -Wall -Wextra -Werror -fstack-usage)
target_link_options(c4_l4_mouse_passthrough PRIVATE -Wl,--undefined=fusion_build_contract)
target_link_options(c4_l4_mouse_passthrough PRIVATE -Wl,--undefined=fusion_hid_layout_contract)
target_link_options(c4_l4_mouse_passthrough PRIVATE -Wl,--undefined=fusion_bridge_contract)
target_link_options(c4_l4_mouse_passthrough PRIVATE -Wl,--undefined=fusion_lifecycle_contract)
target_link_options(c4_l4_mouse_passthrough PRIVATE -Wl,--undefined=fusion_bringup_contract)
pico_enable_stdio_usb(c4_l4_mouse_passthrough 0)
pico_enable_stdio_uart(c4_l4_mouse_passthrough 0)
if(FUSION_TRAJECTORY)
    pico_set_program_name(c4_l4_mouse_passthrough "C4-L4 Offline Preset Candidate")
    if(FUSION_PERSISTENCE)
        pico_set_program_version(c4_l4_mouse_passthrough "0.6.3-ai-smooth")
    else()
        pico_set_program_version(c4_l4_mouse_passthrough "0.5.1-template")
    endif()
    target_link_options(c4_l4_mouse_passthrough PRIVATE -Wl,--undefined=fusion_bypass_contract -Wl,--undefined=fusion_mode_control_contract)
elseif(FUSION_J4_AUX)
    pico_set_program_name(c4_l4_mouse_passthrough "C4-L4 Standalone Open API")
    pico_set_program_version(c4_l4_mouse_passthrough "0.3.0-api")
elseif(FUSION_J4_LINK)
    pico_set_program_name(c4_l4_mouse_passthrough "C4-L4 Standalone HID J4")
    pico_set_program_version(c4_l4_mouse_passthrough "0.2.0-j4-proto")
elseif(FUSION_STANDALONE_FLASH)
    pico_set_program_name(c4_l4_mouse_passthrough "C4-L4 Standalone HID Mouse")
    pico_set_program_version(c4_l4_mouse_passthrough "0.1.0-standalone-candidate")
elseif(FUSION_CONTROLLED_J2)
    pico_set_program_name(c4_l4_mouse_passthrough "C4-L4 Controlled J2 RAM")
    pico_set_program_version(c4_l4_mouse_passthrough "0.1.0-j2-controlled-test")
else()
    pico_set_program_name(c4_l4_mouse_passthrough "C4-L4 Task 4 RAM Bridge")
    pico_set_program_version(c4_l4_mouse_passthrough "0.1.0-task4-private-test")
endif()
pico_add_extra_outputs(c4_l4_mouse_passthrough)
# Compile the actual native-test sources for API/type consistency. These objects are NOT run or linked into firmware.
add_library(task1_test_compile OBJECT tests/test_buttons.cpp tests/test_log_queue.cpp tests/test_hid_input.cpp
    tests/test_hid.cpp tests/test_hid_reject.cpp tests/test_hid_session.cpp tests/test_hid_selection.cpp)
target_include_directories(task1_test_compile PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(task1_test_compile PRIVATE -Wall -Wextra -Werror -O0)
